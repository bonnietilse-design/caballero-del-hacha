const AUDIO_RATE = 22050;
let actx = null, master = null, soundOn = true;
const sfx = {};

function synth(rate, dur, f1, f2 = null, wave = "square", vol = 0.4) {
  if (f2 === null) f2 = f1;
  const n = Math.floor(rate * dur), out = new Float32Array(n);
  let phase = 0;
  for (let i = 0; i < n; i++) {
    const t = i / n;
    phase += (f1 + (f2 - f1) * t) / rate;
    const p = phase % 1;
    let v = wave === "square" ? (p < 0.5 ? 1 : -1) : wave === "tri" ? 4 * Math.abs(p - 0.5) - 1 : Math.sin(2 * Math.PI * p);
    out[i] = v * (1 - t) * vol;
  }
  return out;
}

function buildSounds(rate) {
  return {

    hit: synth(rate, 0.12, 180, 50, "sine", 0.8),

    coin: synth(rate, 0.15, 988, 1319, "square", 0.15),
    hurt: synth(rate, 0.25, 400, 120, "square", 0.3),
    win: synth(rate, 0.4, 400, 800, "tri", 0.4)
  };
}

function initAudio() {
  if (actx) { if (actx.state === "suspended") actx.resume(); return; }
  const AC = window.AudioContext || window.webkitAudioContext;
  if (!AC) return;
  try { actx = new AC(); } catch (e) { return; }
  master = actx.createGain(); master.gain.value = soundOn ? 0.8 : 0; master.connect(actx.destination);
  setTimeout(() => {
    try {
      for (const [name, data] of Object.entries(buildSounds(AUDIO_RATE))) {
        const buf = actx.createBuffer(1, data.length, AUDIO_RATE), ch = buf.getChannelData(0);
        for (let i = 0; i < data.length; i++) ch[i] = Math.max(-1, Math.min(1, data[i]));
        sfx[name] = buf;
      }
    } catch (e) {}
  }, 0);
}

function play(name) {
  if (!soundOn || !actx || !sfx[name]) return;
  const src = actx.createBufferSource(); src.buffer = sfx[name]; src.connect(master); src.start();
}

// ------------------------------------------------------------------ physics
function physicsMove(ent, level) {
  ent.hitWall = false;
  ent.x += ent.vx;
  let r = ent.rect();
  for (const t of level.tilesNear(r)) {
    if (!collide(r, t)) continue;
    if (ent.vx > 0) ent.x = t.x - ent.w; else if (ent.vx < 0) ent.x = t.x + TILE; else continue;
    ent.hitWall = true; ent.vx = 0; r = ent.rect();
  }
  if (level.mode !== "plane") {
    ent.vy = Math.min(ent.vy + GRAVITY, MAX_FALL);
    ent.y += ent.vy;
    r = ent.rect();
    ent.onGround = false;
    for (const t of level.tilesNear(r)) {
      if (!collide(r, t)) continue;
      if (ent.vy > 0) { ent.y = t.y - ent.h; ent.onGround = true; } else if (ent.vy < 0) ent.y = t.y + TILE;
      ent.vy = 0; r = ent.rect();
    }
    if (!ent.onGround && ent.vy >= 0) {
      const probe = { x: r.x, y: r.y + 1, w: r.w, h: r.h };
      if (level.tilesNear(probe).some(t => collide(probe, t))) ent.onGround = true;
    }
  } else {
    ent.y += ent.vy;
  }
}

// ------------------------------------------------------------------ player
class Player {
  constructor(x, y, mode = "standard") {
    this.x = x; this.y = y; this.w = KNIGHT_W; this.h = KNIGHT_H; this.vx = 0; this.vy = 0; this.facing = 1;
    this.onGround = false; this.hitWall = false; this.invuln = 0; this.knockback = 0;
    this.maxHearts = MAX_HEARTS; this.hearts = this.maxHearts;
    this.atkTotal = 22; this.atkCooldown = 6;
    this.attackTimer = 0; this.cooldown = 0; this.coyote = 0; this.jumpBuffer = 0;
    this.hitIds = new Set(); this.dead = false; this.justJumped = false; this.justSwung = false; this.justFired = false;
    this.mode = mode;
  }
  rect() { return { x: Math.round(this.x), y: Math.round(this.y), w: this.w, h: this.h }; }
  swingActive() { const f = this.attackTimer ? this.atkTotal - this.attackTimer : 0; return f >= 6 && f <= 14; }
  attackBox() {
    if (!this.swingActive() || this.mode === "plane" || this.mode === "ninja") return null;
    const r = this.rect(), w = this.mode === "horse" ? 110 : 85, cx = r.x + Math.floor(r.w / 2);
    return { x: this.facing > 0 ? cx : cx - w, y: r.y - 20, w, h: r.h + 40 };
  }
  hurt(srcX) {
    if (this.invuln > 0 || this.dead) return false;
    this.hearts--; this.invuln = INVULN_FRAMES; this.knockback = 12;
    this.vx = srcX > this.x + this.w / 2 ? -6 : 6; this.vy = -6;
    return true;
  }
  update(inp, level) {
    this.justJumped = this.justSwung = this.justFired = false;
    if (this.invuln > 0) this.invuln--;
    if (this.cooldown > 0) this.cooldown--;

    if (inp.attack && this.cooldown === 0) {
      if (this.mode === "plane" || this.mode === "ninja") {
        this.justFired = true;
        this.cooldown = this.mode === "plane" ? 5 : 12;
      } else if (this.attackTimer === 0) {
        this.attackTimer = this.atkTotal; this.hitIds.clear(); this.justSwung = true;
      }
    }
    if (this.attackTimer > 0) { this.attackTimer--; if (this.attackTimer === 0) this.cooldown = this.atkCooldown; }

    let moveX = (inp.right ? 1 : 0) - (inp.left ? 1 : 0);

    if (this.mode === "plane") {
      this.vx = moveX * 5 + 3;
      if (inp.left) this.vx = 1;
      if (inp.right) this.vx = 7;
      if (inp.jumpHeld) this.vy = -5;
      else this.vy = 3;
    } else {
      let maxSpeed = this.mode === "ninja" ? 6.8 : this.mode === "horse" ? 7.8 : MAX_RUN;
      let accel = (this.onGround ? RUN_ACCEL : RUN_ACCEL * AIR_CONTROL);
      if (moveX) {
        this.vx = Math.max(-maxSpeed, Math.min(maxSpeed, this.vx + moveX * accel));
        if (this.attackTimer === 0) this.facing = moveX;
      } else if (this.onGround && this.knockback === 0) {
        this.vx *= RUN_FRICTION; if (Math.abs(this.vx) < 0.1) this.vx = 0;
      }

      this.coyote = this.onGround ? COYOTE_FRAMES : Math.max(0, this.coyote - 1);
      this.jumpBuffer = inp.jumpPressed ? JUMP_BUFFER_FRAMES : Math.max(0, this.jumpBuffer - 1);
      
      if (this.mode === "ninja" && this.hitWall && !this.onGround && inp.jumpPressed) {
        this.vy = -JUMP_SPEED;
        this.vx = -this.facing * 7;
        this.facing = -this.facing;
        this.justJumped = true;
      } else if (this.jumpBuffer && this.coyote) {
        this.vy = -JUMP_SPEED; this.jumpBuffer = this.coyote = 0; this.onGround = false; this.justJumped = true;
        play("jump");
      }
      if (!inp.jumpHeld && this.vy < -JUMP_CUT) this.vy = -JUMP_CUT;
    }

    physicsMove(this, level);
    if (this.x < 0) { this.x = 0; this.vx = 0; }
    if (this.y < 0) this.y = 0;
    if (this.y > level.pxH - this.h && this.mode === "plane") this.y = level.pxH - this.h;
    this.x = Math.min(this.x, level.pxW - this.w);
  }
}

// ------------------------------------------------------------------ enemies
let nextId = 1;
class Enemy {
  init(kind, w, h, hp, score) {
    Object.assign(this, { kind, w, h, hp, maxHp: hp, score, id: nextId++, vx: 0, vy: 0, dir: -1,
      onGround: false, hitWall: false, flash: 0, stun: 0, alive: true, t: Math.random() * 100 });
  }
  rect() { return { x: Math.round(this.x), y: Math.round(this.y), w: this.w, h: this.h }; }
  hit(fromDir, dmg = 1) {
    this.hp -= dmg; this.flash = 8; this.stun = 16; this.vx = fromDir * 6.5; this.vy = -4; this.dir = -fromDir;
    if (this.hp <= 0) { this.alive = false; return true; }
    return false;
  }
  harmful() { return this.stun === 0; }
  hittable() { return true; }
}

class Kangaroo extends Enemy {
  constructor(kind, tx, ty) {
    super();
    const hp = kind === "T" ? 3 : 1;
    this.init(kind, 30, 42, hp, kind === "T" ? 500 : 100);
    this.x = tx; this.y = ty; this.wait = randInt(10, 40);
  }
  update(level) {
    this.t++; if (this.flash) this.flash--;
    if (this.onGround && --this.wait <= 0) {
      this.vy = -8; this.vx = this.dir * 3; this.wait = randInt(30, 60);
    }
    physicsMove(this, level);
    if (this.hitWall) this.dir *= -1;
  }
  draw(cam) {
    const r = this.rect();
    P(ENEMY_COLORS[this.kind][0], [[r.x - cam, r.y + r.h], [r.x - cam + r.w / 2, r.y], [r.x - cam + r.w, r.y + r.h]]);
  }
}

class NinjaEnemy extends Enemy {
  constructor(tx, ty) {
    super(); this.init("N", 30, 48, 2, 250);
    this.x = tx; this.y = ty;
  }
  update(level, player, game) {
    this.t++; if (this.flash) this.flash--;
    const dx = player.x - this.x;
    this.dir = dx > 0 ? 1 : -1;
    if (Math.abs(dx) < 250 && Math.random() < 0.02) {
      game.shots.push({ kind: "shuriken", enemy: true, x: this.x, y: this.y + 15, vx: this.dir * 7, vy: 0, t: 0 });
    }
    physicsMove(this, level);
  }
  draw(cam) {
    const r = this.rect();
    R(BLACK, r.x - cam, r.y, r.w, r.h, 4);
    R(rgb(200, 30, 30), r.x - cam + (this.dir > 0 ? 15 : 5), r.y + 10, 10, 4);
  }
}

// ------------------------------------------------------------------ level
class Level {
  constructor(def) {
    this.name = def.name; this.hint = def.hint || ""; this.theme = THEMES[def.theme];
    this.mode = def.mode || "standard";
    this.rows = def.map.map(r => r.padEnd(Math.max(...def.map.map(x => x.length)), "."));
    this.w = this.rows[0].length; this.h = this.rows.length;
    this.pxW = this.w * TILE; this.pxH = this.h * TILE;
    this.tiles = new Map(); this.coins = []; this.enemies = [];
    this.start = [TILE * 2, TILE * 9]; this.flag = null; this.boss = null;

    this.rows.forEach((row, cy) => [...row].forEach((ch, cx) => {
      const x = cx * TILE, y = cy * TILE;
      if (SOLID.includes(ch)) this.tiles.set(cx + "," + cy, ch);
      else if (ch === "o") this.coins.push({ x: x + 20, y: y + 20, t: Math.random() * 100 });
      else if (ch === "P") this.start = [x, y];
      else if (ch === "t" || ch === "T" || ch === "j") this.enemies.push(new Kangaroo(ch, x, y));
      else if (ch === "N") this.enemies.push(new NinjaEnemy(x, y));
      else if (ch === "E") this.enemies.push(new PlaneEnemy(x, y));
      else if (ch === "J") this.enemies.push(new JoustEnemy(x, y));
      else if (ch === "X") { this.boss = new BlackKnightBoss(x, y); this.enemies.push(this.boss); }
      else if (ch === "F") this.flag = { x: x + 15, y: y - TILE * 6, w: 10, h: TILE * 7 };
    }));
  }
  solidAt(cx, cy) { return this.tiles.has(cx + "," + cy); }
  tilesNear(r) {
    const out = [];
    const x0 = Math.max(Math.floor(r.x / TILE) - 1, 0), x1 = Math.min(Math.floor((r.x + r.w) / TILE) + 1, this.w - 1);
    const y0 = Math.max(Math.floor(r.y / TILE) - 1, 0), y1 = Math.min(Math.floor((r.y + r.h) / TILE) + 1, this.h - 1);
    for (let cy = y0; cy <= y1; cy++) for (let cx = x0; cx <= x1; cx++)
      if (this.solidAt(cx, cy)) out.push({ cx, cy, x: cx * TILE, y: cy * TILE, w: TILE, h: TILE });
    return out;
  }
}

function R(col, x, y, w, h, r = 0) {
  ctx.fillStyle = col;
  if (r && ctx.roundRect) { ctx.beginPath(); ctx.roundRect(x, y, w, h, r); ctx.fill(); } else ctx.fillRect(x, y, w, h);

function P(col, pts) {
  ctx.fillStyle = col; ctx.beginPath(); ctx.moveTo(pts[0][0], pts[0][1]);
  for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i][0], pts[i][1]);

}

}

class Game {
  constructor() {
    this.t = 0; this.state = "title"; this.levelIndex = 0;
    this.lives = START_LIVES; this.score = 0; this.coins = 0;
    this.shots = [];
    this.resetLevel();
  }
  newGame(idx = 0) {
    this.levelIndex = idx; if (idx === 0) { this.score = 0; this.coins = 0; }
    this.lives = START_LIVES; this.resetLevel(); this.state = "play";
    say("vamos", true);
  }
  resetLevel() {
    this.level = new Level(LEVELS[this.levelIndex]);
    this.player = new Player(...this.level.start, this.level.mode);
    this.camX = 0; this.shots = [];
  }
  step() {
    this.t++;
    if (this.state !== "play") return;
    const p = this.player, lvl = this.level;
    p.update(inp, lvl);

    if (p.justFired) {
      if (p.mode === "ninja") {
        this.shots.push({ kind: "shuriken", enemy: false, x: p.x + (p.facing > 0 ? 30 : -10), y: p.y + 20, vx: p.facing * 10, vy: 0, t: 0 });
        play("shoot");
      } else if (p.mode === "plane") {
        this.shots.push({ kind: "bullet", enemy: false, x: p.x + 50, y: p.y + 16, vx: 14, vy: (Math.random() - 0.5) * 2, t: 0 });
        play("shoot");
      }
    }

    const box = p.attackBox();
    for (const e of lvl.enemies) {
      e.update(lvl, p, this);
      if (box && collide(box, e.rect())) e.hit(p.facing, 2);
      if (e.harmful() && collide(p.rect(), e.rect())) {
        if (p.hurt(e.x)) play("hurt");
      }
    }
    lvl.enemies = lvl.enemies.filter(e => e.alive);

    for (const s of this.shots) {
      s.x += s.vx; s.y += s.vy; s.t++;
      const sr = { x: s.x - 5, y: s.y - 5, w: 10, h: 10 };
      if (s.enemy) {
        if (collide(sr, p.rect())) { p.hurt(s.x); s.dead = true; }
      } else {
        for (const e of lvl.enemies) {
          if (collide(sr, e.rect())) { e.hit(1, 1); s.dead = true; break; }
        }
      }
      if (s.t > 120) s.dead = true;
    }
    this.shots = this.shots.filter(s => !s.dead);

    for (const c of lvl.coins) {
      if (!c.taken && collide(p.rect(), { x: c.x - 10, y: c.y - 10, w: 20, h: 20 })) {
        c.taken = true; this.coins++; this.score += 50; play("coin");
      }
    }

    if (lvl.flag && collide(p.rect(), lvl.flag)) {
      if (this.levelIndex < LEVELS.length - 1) {
        this.levelIndex++; this.resetLevel(); say("nivel", true);
      } else {
        this.state = "win"; play("win"); say("victoria", true);
      }
    }

    if (p.hearts <= 0 || p.y > lvl.pxH + 50) {
      this.lives--;
      if (this.lives > 0) this.resetLevel();
      else { this.state = "gameover"; say("fin", true); }
    }

    this.camX += (p.x - W * 0.35 - this.camX) * 0.1;
    this.camX = Math.max(0, Math.min(this.camX, lvl.pxW - W));
  }
  draw() {
    const cam = Math.floor(this.camX), lvl = this.level;
    ctx.fillStyle = lvl.theme.sky[0][0] ? rgb(...lvl.theme.sky[0]) : BLACK;
    ctx.fillRect(0, 0, W, H);

    for (let cx = Math.floor(cam / TILE); cx <= Math.floor((cam + W) / TILE) + 1; cx++) {
      for (let cy = 0; cy < lvl.h; cy++) {
        if (lvl.solidAt(cx, cy)) R(lvl.theme.top[0], cx * TILE - cam, cy * TILE, TILE, TILE);
      }
    }

    for (const c of lvl.coins) {
      if (!c.taken) R(GOLD, c.x - cam - 8, c.y - 8, 16, 16, 8);
    }

    if (lvl.flag) R(rgb(220, 40, 40), lvl.flag.x - cam, lvl.flag.y, lvl.flag.w, lvl.flag.h);

    for (const e of lvl.enemies) e.draw(cam);
    drawPlayer(this.player, cam);

    for (const s of this.shots) {
      if (s.kind === "shuriken") R(SILVER, s.x - cam - 4, s.y - 4, 8, 8, 2);
      else R(GOLD, s.x - cam - 3, s.y - 3, 6, 6, 3);
    }

    T(`VIDAS ${this.lives}  MONEDAS ${this.coins}  PUNTOS ${this.score}`, 20, 20, 30, WHITE, "left");
    T(`NIVEL ${this.levelIndex + 1}: ${lvl.name}`, 20, W - 20, 30, GOLD, "right");

    if (this.state === "title") {
      R("rgba(0,0,0,0.75)", 0, 0, W, H);
      T("EL CABALLERO DEL HACHA", 50, W / 2, 180, GOLD);
      T("¡Edición de Batalla Multimodal!", 24, W / 2, 240, WHITE);
      T("Pulsa ENTER o Toca para Empezar", 26, W / 2, 340, GRASS_LIGHT);
    } else if (this.state === "gameover") {
      R("rgba(0,0,0,0.8)", 0, 0, W, H);
      T("FIN DEL JUEGO", 60, W / 2, 220, HEART);
      T("Pulsa ENTER para Reintentar", 26, W / 2, 320, WHITE);
    } else if (this.state === "win") {
      R("rgba(0,0,0,0.8)", 0, 0, W, H);
      T("¡VICTORIA TOTAL!", 60, W / 2, 200, GOLD);
      T(`Puntuación Final: ${this.score}`, 30, W / 2, 280, WHITE);
      T("¡Has completado todas las misiones!", 22, W / 2, 340, GRASS_LIGHT);
    }
  }
}
